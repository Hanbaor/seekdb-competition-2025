```cpp
// @/home/tian/seekdb/src/sql/das/iter/ob_das_text_retrieval_iter.cpp#372-409

int ObDASTextRetrievalIter::inner_get_next_row()
{
  int ret = OB_SUCCESS;
  if (IS_NOT_INIT) {
    ret = OB_NOT_INIT;
    LOG_WARN("retrieval iterator not inited", K(ret));
  } else if (!inv_idx_agg_evaluated_ && need_inv_idx_agg_) {
    // 聚合操作（计算文档总数等）
    if (OB_FAIL(do_doc_cnt_agg())) {
      // ...
    } else {
      inv_idx_agg_evaluated_ = true;
    }
  }

  if (OB_FAIL(ret)) {
  } 
  // ========== 关键步骤：从倒排索引读取 posting ==========
  else if (OB_FAIL(inverted_idx_scan_iter_->get_next_row())) {
    if (OB_UNLIKELY(OB_ITER_END != ret)) {
      LOG_WARN("failed to get next row from inverted index", K(ret));
    }
  } 
  // ========== 成功读取一行后的处理 ==========
  else {
    LOG_DEBUG("get one invert index scan row", "row",
        ROWEXPR2STR(*ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_,
        *inv_idx_scan_param_.output_exprs_));
    
    // 如果需要计算相关性分数
    if (ir_ctdef_->need_calc_relevance()) {
      clear_row_wise_evaluated_flag();
      
      // 获取文档的 token 数量（用于 BM25 计算）
      if (OB_FAIL(get_next_doc_token_cnt(need_fwd_idx_agg_))) {
        LOG_WARN("failed to get next doc token count", K(ret));
      } 
      // 填充 token 和文档计数
      else if (OB_FAIL(fill_token_doc_cnt())) {
        LOG_WARN("failed to get token doc cnt", K(ret));
      } 
      // 计算相关性分数
      else if (OB_FAIL(project_relevance_expr())) {
        LOG_WARN("failed to evaluate simarity expr", K(ret));
      }
    }
  }
  return ret;
}
```

### 1.2 当前代码的执行流程

```
inner_get_next_row()
  ↓
inverted_idx_scan_iter_->get_next_row()  ← 读取倒排索引的一个 posting
  ↓
读取到的字段（通过 output_exprs_）：
  - doc_id (ir_ctdef_->inv_scan_domain_id_col_)
  - token 相关信息
  - 可能的位置信息
  ↓
计算相关性分数（如果需要）：
  - get_next_doc_token_cnt() ← 可能需要查询正排索引
  - fill_token_doc_cnt()
  - project_relevance_expr() ← 计算 BM25 等
  ↓
返回 OB_SUCCESS
  ↓
❌ 没有标量过滤！
  ↓
数据返回到上层（ObDASFuncDataIter 等）
  ↓
上层需要回表查询 id, base_id 进行过滤  ← 性能瓶颈！
```

### 1.3 关键观察

**当前代码只读取了**：
- ✅ `doc_id`（文档ID）
- ✅ `relevance`（相关性分数）
- ✅ token 相关信息（词频、位置等）

**当前代码没有读取**：
- ❌ `id`（主表的 id 字段）
- ❌ `base_id`（主表的 base_id 字段）

**原因**：倒排索引的 Posting 结构中**没有存储**这些标量字段！

## 2. 问题根源：倒排索引没有嵌入标量字段

### 2.1 当前倒排索引的 Posting 结构

让我们看看实际存储的是什么：

```cpp
// 当前倒排索引存储的 Posting（推测）
struct CurrentPosting {
  uint64_t doc_id;        // 文档ID
  float relevance;        // 相关性（可能在查询时计算）
  uint32_t term_freq;     // 词频
  uint32_t positions[];   // 位置信息（可选）
  // ❌ 没有 cached_id
  // ❌ 没有 cached_base_id
};
```

**物理存储示例**：
```
Token: "database"
Posting List (磁盘文件):
  [doc_001, term_freq: 2, positions: [5, 20]]
  [doc_002, term_freq: 1, positions: [10]]
  [doc_003, term_freq: 3, positions: [2, 15, 30]]
  
❌ 没有 id 和 base_id 信息！
```

### 2.2 这就是为什么需要"嵌入"

如果倒排索引中**没有存储标量字段**，那么：
1. ❌ 无法在 `inner_get_next_row()` 中读取 `id`, `base_id`
2. ❌ 无法在这里进行过滤
3. ❌ 必须返回到上层，回表查询主表

```cpp
  float relevance;           // 4 bytes - 现有字段（或查询时计算）
```

```cpp
// 索引构建代码（需要修改的位置）

int build_fulltext_index(...)
{
  // 扫描主表
  while (OB_SUCC(scan_main_table())) {
    // 读取所有需要的字段
    ObString fulltext = get_fulltext_column();
    uint64_t id = get_id_column();              // ← 新增：读取 id
    ObString base_id = get_base_id_column();    // ← 新增：读取 base_id
    
    // 分词

    tokenize(fulltext, tokens);
    
    // 为每个 token 创建 posting
    for (int i = 0; i < tokens.count(); ++i) {
      EnhancedPosting posting;
      posting.doc_id = current_doc_id;
      posting.term_freq = calculate_term_freq(tokens[i], fulltext);
      
      // ========== 嵌入标量字段 ==========

      posting.cached_base_id = hash(base_id);      // ← 新增
      
      // 写入倒排索引
      write_posting(tokens[i], posting);
    }
  }
}
```

### 第二步：修改查询代码，读取并使用嵌入的字段

#### 3.4 修改 `ObDASTextRetrievalIter::inner_get_next_row()`

```cpp
// @/home/tian/seekdb/src/sql/das/iter/ob_das_text_retrieval_iter.cpp

int ObDASTextRetrievalIter::inner_get_next_row()
{
  int ret = OB_SUCCESS;
  
  // ========== 原有代码保持不变 ==========
  if (IS_NOT_INIT) {
    ret = OB_NOT_INIT;
    LOG_WARN("retrieval iterator not inited", K(ret));
  } else if (!inv_idx_agg_evaluated_ && need_inv_idx_agg_) {
    if (OB_FAIL(do_doc_cnt_agg())) {
      // ...
    } else {
      inv_idx_agg_evaluated_ = true;
    }
  }

  // ========== 修改：使用 while 循环实现边扫描边过滤 ==========
  while (OB_SUCC(ret)) {
    // 从倒排索引读取下一行
    if (OB_FAIL(inverted_idx_scan_iter_->get_next_row())) {
      if (OB_UNLIKELY(OB_ITER_END != ret)) {
        LOG_WARN("failed to get next row from inverted index", K(ret));
      }
      break;  // 结束循环
    }
    
    // ========== 新增：读取嵌入的标量字段 ==========
    ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;
    
    // 读取嵌入的 id 字段
    uint64_t cached_id = 0;
    if (ir_ctdef_->cached_id_col_ != nullptr) {  // ← 需要在 ctdef 中添加
      cached_id = ir_ctdef_->cached_id_col_
                  ->locate_expr_datum(*eval_ctx).get_uint64();
    }
    
    // 读取嵌入的 base_id 字段
    uint32_t cached_base_id = 0;
    if (ir_ctdef_->cached_base_id_col_ != nullptr) {  // ← 需要在 ctdef 中添加
      cached_base_id = ir_ctdef_->cached_base_id_col_
                       ->locate_expr_datum(*eval_ctx).get_uint32();
    }
    
    // ========== 新增：应用标量过滤 ==========
    // 检查 id 过滤条件
    if (ir_ctdef_->has_id_filter_ && cached_id >= ir_ctdef_->filter_max_id_) {
      // 不满足条件，继续循环读取下一个 posting
      continue;
    }
    
    // 检查 base_id 过滤条件
    if (ir_ctdef_->has_base_id_filter_) {
      bool match = false;
      for (int i = 0; i < ir_ctdef_->filter_base_id_hashes_.count(); ++i) {
        if (cached_base_id == ir_ctdef_->filter_base_id_hashes_[i]) {
          match = true;
          break;
        }
      }
      if (!match) {
        // 不满足条件，继续循环
        continue;
      }
    }
    
    // ========== 通过过滤，执行原有逻辑 ==========
    LOG_DEBUG("get one invert index scan row (after filtering)", "row",
        ROWEXPR2STR(*eval_ctx, *inv_idx_scan_param_.output_exprs_));
    
    if (ir_ctdef_->need_calc_relevance()) {
      clear_row_wise_evaluated_flag();
      if (OB_FAIL(get_next_doc_token_cnt(need_fwd_idx_agg_))) {
        LOG_WARN("failed to get next doc token count", K(ret));
      } else if (OB_FAIL(fill_token_doc_cnt())) {
        LOG_WARN("failed to get token doc cnt", K(ret));
      } else if (OB_FAIL(project_relevance_expr())) {
        LOG_WARN("failed to evaluate simarity expr", K(ret));
      }
    }
    
    // 成功返回一行（已过滤）
    return OB_SUCCESS;
  }
  
  return ret;  // OB_ITER_END 或错误
}
```

## 7. 总结

### 7.1 我之前给的"简化示例"

我之前给的代码是**理想状态下应该怎么写**，是一个**改造目标**，而不是当前实际代码。

### 7.2 当前实际情况

当前代码：
- ✅ 能从倒排索引读取 `doc_id` 和计算 `relevance`
- ❌ **没有读取标量字段**（因为索引中根本没存）
- ❌ **没有在这里过滤**（需要在上层回表后才能过滤）

### 7.3 改造的核心

1. **第一步（索引构建）**：在 Posting 中**嵌入** `id`, `base_id`
2. **第二步（查询执行）**：在 `inner_get_next_row()` 中**读取并使用**这些字段进行过滤

这就是"嵌入倒排索引"的完整含义！
