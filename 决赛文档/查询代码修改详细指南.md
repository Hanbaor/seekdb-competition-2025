## 1. 现有代码如何读取字段（学习模板）

### 1.1 现有的 doc_id 读取方式

```cpp
// @/home/tian/seekdb/src/sql/das/iter/ob_das_text_retrieval_iter.cpp#630-641

int ObDASTextRetrievalIter::get_inv_idx_scan_doc_id(ObDocIdExt &doc_id)
{
  int ret = OB_SUCCESS;
  
  // 第1步：从 ctdef 获取字段表达式
  sql::ObExpr *doc_id_expr = ir_ctdef_->inv_scan_domain_id_col_;
  
  // 第2步：从 rtdef 获取执行上下文
  sql::ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;
  
  // 第3步：定位到 Datum（数据）
  ObDatum &doc_id_datum = doc_id_expr->locate_expr_datum(*eval_ctx);
  
  // 第4步：从 Datum 中提取值
  if (OB_FAIL(doc_id.from_datum(doc_id_datum))) {
    LOG_WARN("failed to get doc id", K(ret), K(doc_id_datum));
  }

  return ret;
}
```

**这个模式告诉我们**：
1. 字段表达式存储在 `ir_ctdef_` 中
2. 执行上下文存储在 `ir_rtdef_` 中
3. 使用 `locate_expr_datum()` 获取数据
4. 使用对应的方法提取具体类型的值

// 读取嵌入的 id 字段
int ObDASTextRetrievalIter::get_cached_id(uint64_t &cached_id)
{
  int ret = OB_SUCCESS;
  cached_id = UINT64_MAX;  // 默认值
  
  // 检查是否有嵌入的 id 字段
  if (ir_ctdef_->cached_id_col_ == nullptr) {
    // 没有嵌入字段，返回默认值
    return OB_SUCCESS;
  }
  
  // 第1步：获取字段表达式
  sql::ObExpr *cached_id_expr = ir_ctdef_->cached_id_col_;
  
  // 第2步：获取执行上下文
  sql::ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;
  
  // 第3步：定位到 Datum
  ObDatum &cached_id_datum = cached_id_expr->locate_expr_datum(*eval_ctx);
  
  // 第4步：从 Datum 中提取 uint64_t 值
  if (cached_id_datum.is_null()) {
    // NULL 值，使用默认值
    cached_id = UINT64_MAX;
  } else {
    cached_id = cached_id_datum.get_uint64();
  }
  
  return ret;
}

// 读取嵌入的 base_id 字段（哈希值）
int ObDASTextRetrievalIter::get_cached_base_id(uint32_t &cached_base_id_hash)
{
  int ret = OB_SUCCESS;
  cached_base_id_hash = 0;  // 默认值
  
  // 检查是否有嵌入的 base_id 字段
  if (ir_ctdef_->cached_base_id_col_ == nullptr) {
    return OB_SUCCESS;
  }
  
  sql::ObExpr *cached_base_id_expr = ir_ctdef_->cached_base_id_col_;
  sql::ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;
  ObDatum &cached_base_id_datum = cached_base_id_expr->locate_expr_datum(*eval_ctx);
  
  if (cached_base_id_datum.is_null()) {
    cached_base_id_hash = 0;
  } else {
    // 假设存储的是哈希值（uint32_t）
    cached_base_id_hash = cached_base_id_datum.get_uint32();
  }
  
  return ret;
}

// 检查是否满足标量过滤条件
bool ObDASTextRetrievalIter::check_scalar_filter(
    uint64_t cached_id, 
    uint32_t cached_base_id_hash)
{
  // 如果没有过滤条件，全部通过
  if (!ir_ctdef_->has_scalar_filter_) {
    return true;
  }
  
  // 检查 id 过滤条件
  if (cached_id >= ir_ctdef_->filter_max_id_) {
    return false;  // 不满足
  }
  
  // 检查 base_id 过滤条件
  if (ir_ctdef_->filter_base_id_hashes_.count() > 0) {
    bool match = false;
    for (int64_t i = 0; i < ir_ctdef_->filter_base_id_hashes_.count(); ++i) {
      if (cached_base_id_hash == ir_ctdef_->filter_base_id_hashes_.at(i)) {
        match = true;
        break;
      }
    }
    if (!match) {
      return false;  // 不满足
    }
  }
  
  return true;  // 通过所有过滤条件
}

### 3.2 修改后的代码（边扫描边过滤）

```cpp
int ObDASTextRetrievalIter::inner_get_next_row()
{
  int ret = OB_SUCCESS;
  
  // ========== 原有的初始化检查保持不变 ==========
  if (IS_NOT_INIT) {
    ret = OB_NOT_INIT;
    LOG_WARN("retrieval iterator not inited", K(ret));
  } else if (!inv_idx_agg_evaluated_ && need_inv_idx_agg_) {
    if (OB_FAIL(do_doc_cnt_agg())) {
      if (OB_UNLIKELY(OB_ITER_END != ret)) {
        LOG_WARN("Fail to do document count aggregation", K(ret));
      }
    } else {
      inv_idx_agg_evaluated_ = true;
    }
  }

  // ========== 修改：使用 while 循环实现边扫描边过滤 ==========
  while (OB_SUCC(ret)) {
    // 从倒排索引读取下一行
    if (OB_FAIL(inverted_idx_scan_iter_->get_next_row())) {
      if (OB_UNLIKELY(OB_ITER_END != ret)) {
        LOG_WARN("failed to get next row from inverted index", K(ret));
      }
      break;  // 结束循环
    }
    
    LOG_DEBUG("get one invert index scan row", "row",
        ROWEXPR2STR(*ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_,
        *inv_idx_scan_param_.output_exprs_));
    
    // ========== 新增：读取嵌入的标量字段 ==========
    uint64_t cached_id = UINT64_MAX;
    uint32_t cached_base_id_hash = 0;
    
    if (OB_FAIL(get_cached_id(cached_id))) {
      LOG_WARN("failed to get cached id", K(ret));
      break;
    }
    
    if (OB_FAIL(get_cached_base_id(cached_base_id_hash))) {
      LOG_WARN("failed to get cached base_id", K(ret));
      break;
    }
    
    // ========== 新增：应用标量过滤 ==========
    if (!check_scalar_filter(cached_id, cached_base_id_hash)) {
      // 不满足过滤条件，跳过这一行，继续读取下一行
      LOG_DEBUG("row filtered by scalar condition", 
                K(cached_id), K(cached_base_id_hash));
      continue;  // ← 关键：继续循环，不返回这一行
    }
    
    // ========== 通过过滤，执行原有逻辑 ==========
    LOG_DEBUG("row passed scalar filter", K(cached_id), K(cached_base_id_hash));
    
    if (ir_ctdef_->need_calc_relevance()) {
      clear_row_wise_evaluated_flag();
      if (OB_FAIL(get_next_doc_token_cnt(need_fwd_idx_agg_))) {
        LOG_WARN("failed to get next doc token count", K(ret));
        break;
      } else if (OB_FAIL(fill_token_doc_cnt())) {
        LOG_WARN("failed to get token doc cnt", K(ret));
        break;
      } else if (OB_FAIL(project_relevance_expr())) {
        LOG_WARN("failed to evaluate simarity expr", K(ret));
        break;
      }
    }
    
    // 成功返回一行（已过滤）
    return OB_SUCCESS;
  }
  
  return ret;  // OB_ITER_END 或错误
}
```

## 4. 批量版本：inner_get_next_rows()

批量版本更复杂，需要处理向量化执行：

```cpp
int ObDASTextRetrievalIter::inner_get_next_rows(int64_t &count, int64_t capacity)
{
  int ret = OB_SUCCESS;
  
  // ... 原有的初始化检查 ...
  
  count = 0;
  int64_t real_capacity = OB_MIN(max_batch_size_, capacity);
  
  if (OB_FAIL(ret)) {
  } else if (max_batch_size_ == 0) {
    ret = OB_ERR_UNEXPECTED;
    LOG_WARN("max batch size is 0", K(ret));
  } 
  
  // ========== 修改：读取一批数据后进行过滤 ==========
  else {
    int64_t read_count = 0;
    
    // 读取一批 posting
    if (OB_FAIL(inverted_idx_scan_iter_->get_next_rows(read_count, real_capacity))) {
      if (OB_UNLIKELY(OB_ITER_END != ret)) {
        LOG_WARN("failed to get batch rows from inverted index", K(ret));
      } else if (read_count != 0) {
        ret = OB_SUCCESS;
      }
    }
    
    if (OB_SUCC(ret) && read_count > 0) {
      sql::ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;
      
      // ========== 新增：获取嵌入字段的向量 ==========
      const ObDatumVector *cached_id_vector = nullptr;
      const ObDatumVector *cached_base_id_vector = nullptr;
      
      if (ir_ctdef_->cached_id_col_ != nullptr) {
        cached_id_vector = &(ir_ctdef_->cached_id_col_
                             ->locate_expr_datumvector(*eval_ctx));
      }
      
      if (ir_ctdef_->cached_base_id_col_ != nullptr) {
        cached_base_id_vector = &(ir_ctdef_->cached_base_id_col_
                                   ->locate_expr_datumvector(*eval_ctx));
      }
      
      // ========== 新增：使用 skip 向量标记要过滤的行 ==========
      ObBitVector *skip = nullptr;
      if (ir_ctdef_->has_scalar_filter_) {
        // 分配 skip 向量
        void *buf = eval_ctx->exec_ctx_.get_allocator().alloc(
            ObBitVector::memory_size(read_count));
        skip = to_bit_vector(buf);
        skip->reset(read_count);
        
        // 遍历每一行，检查过滤条件
        for (int64_t i = 0; i < read_count; ++i) {
          uint64_t cached_id = UINT64_MAX;
          uint32_t cached_base_id_hash = 0;
          
          // 提取标量字段值
          if (cached_id_vector != nullptr) {
            cached_id = cached_id_vector->at(i)->get_uint64();
          }
          
          if (cached_base_id_vector != nullptr) {
            cached_base_id_hash = cached_base_id_vector->at(i)->get_uint32();
          }
          
          // 检查过滤条件
          if (!check_scalar_filter(cached_id, cached_base_id_hash)) {
            skip->set(i);  // 标记为跳过
          }
        }
      }
      
      // ========== 原有逻辑：计算相关性（跳过被过滤的行）==========
      if (ir_ctdef_->need_calc_relevance()) {
        PRINT_VECTORIZED_ROWS(SQL, DEBUG, *eval_ctx, 
                              *inv_idx_scan_param_.output_exprs_, 
                              read_count, skip);
        clear_batch_wise_evaluated_flag(read_count);
        
        if (OB_FAIL(batch_fill_token_cnt_with_doc_len(read_count))) {
          LOG_WARN("failed to fill batch token cnt with document length", K(ret));
        } else if (OB_FAIL(fill_token_doc_cnt())) {
          LOG_WARN("failed to get token doc cnt", K(ret));
        } else if (OB_FAIL(batch_project_relevance_expr(read_count))) {
          LOG_WARN("failed to evaluate simarity expr", K(ret));
        }
      }
      
      // ========== 新增：计算实际输出的行数（排除被 skip 的行）==========
      if (OB_SUCC(ret)) {
        count = 0;
        for (int64_t i = 0; i < read_count; ++i) {
          if (skip == nullptr || !skip->at(i)) {
            count++;
          }
        }
        
        LOG_DEBUG("batch filtering result", 
                  K(read_count), K(count), "filtered", read_count - count);
      }
    }
  }
  
  return ret;
}
```

## 6. 执行流程对比

### 6.1 修改前的流程

```
inner_get_next_row()
  ↓
inverted_idx_scan_iter_->get_next_row()
  ↓
读取到 posting {doc_id, relevance, ...}
  ↓
计算相关性分数
  ↓
return OB_SUCCESS (返回所有行，未过滤)
  ↓
上层 (ObDASFuncDataIter) 接收到 doc_id
  ↓
回表查询主表：SELECT id, base_id FROM items1 WHERE docid = ...
  ↓
应用过滤：if (id < 1000 && base_id IN (...))
  ↓
返回最终结果
```

### 6.2 修改后的流程

```
inner_get_next_row()
  ↓
while (true):
  ↓
  inverted_idx_scan_iter_->get_next_row()
  ↓
  读取到 posting {doc_id, relevance, cached_id, cached_base_id, ...}
  ↓
  get_cached_id() → cached_id
  get_cached_base_id() → cached_base_id_hash
  ↓
  check_scalar_filter(cached_id, cached_base_id_hash)?
  ├─ 不满足 → continue (读取下一个 posting)
  └─ 满足 ↓
  计算相关性分数
  ↓
  return OB_SUCCESS (只返回通过过滤的行)
  ↓
上层接收到已过滤的 doc_id
  ↓
无需回表！直接返回结果
```

## 7. 关键点总结

### 7.1 读取字段的核心步骤

```cpp
// 步骤1：从 ctdef 获取表达式
sql::ObExpr *field_expr = ir_ctdef_->cached_xxx_col_;

// 步骤2：从 rtdef 获取上下文
sql::ObEvalCtx *eval_ctx = ir_rtdef_->get_inv_idx_scan_rtdef()->eval_ctx_;

// 步骤3：定位到 Datum
ObDatum &datum = field_expr->locate_expr_datum(*eval_ctx);

// 步骤4：提取值
value = datum.get_xxx();  // get_uint64(), get_uint32(), get_string(), ...
```

### 7.2 过滤逻辑的核心

```cpp
while (读取 posting) {
  // 读取嵌入字段
  cached_values = read_embedded_fields();
  
  // 检查过滤条件
  if (!满足条件(cached_values)) {
    continue;  // 跳过，继续读下一个
  }
  
  // 通过过滤
  return posting;
}
```
